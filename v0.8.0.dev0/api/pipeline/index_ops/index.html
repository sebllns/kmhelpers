    class IndexOps:
    """Orchestrates k-mer index builds against the kmindex registry ``BUILD_DIR``.

    Loads an index definition or a span registry (see ``sources``), builds
    every selected definition as unregistered kmtricks runs, then publishes
    each target: the runs of its parts, merged with its current version when
    it is already registered, become ``BUILD_DIR/<target>``. A definition
    that is not a part of a selected merge is its own target. In plan/dry-run
    modes, commands are only generated and can be exported as shell scripts
    via ``write_script()``.

    Args:
        config: Runtime configuration governing paths, build parameters,
            filtering, and execution behaviour.  See ``IndexOpsConfig``.

    Attributes:
        config (IndexOpsConfig): The resolved configuration (``workdir`` is
            made absolute on construction).
        work_dir (str): ``BUILD_DIR``, the kmindex registry.
        asset_dir (str): ``<work_dir>/assets`` - one folder of scripts per session.
        log_dir (str): ``<work_dir>/logs`` - log file destination.
        timestamp (str): ``YYYYmmdd_HHMMSS`` string captured at construction.
    """

    def __init__(self, config: IndexOpsConfig) -> None:
        self._config = config
        self._config.workdir = os.path.realpath(config.workdir)
        self._timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")

        # Kept across runs: sample files are resolved relative to their definition file
        self._dbs = DbCache()
        self._script = ScriptRecorder(self.work_dir)
        self._paths: Optional[BuildPaths] = None

        logger.debug(f"Init {type(self).__name__}")
        logger.debug("workdir: " + self.work_dir)

        for d in (self.work_dir, self.log_dir, self.asset_dir):
            os.makedirs(d, exist_ok=True)

    # ---
    # PROPERTIES

    @property
    def config(self) -> IndexOpsConfig:
        return self._config

    @property
    def work_dir(self) -> str:
        return self.config.workdir

    @property
    def asset_dir(self) -> str:
        return os.path.join(self.work_dir, "assets")

    @property
    def log_dir(self) -> str:
        return os.path.join(self.work_dir, "logs")

    @property
    def timestamp(self) -> str:
        return self._timestamp

    @property
    def paths(self) -> Optional[BuildPaths]:
        """Layout of the session of the last ``run()``."""
        return self._paths

    # ---
    # PUBLIC METHODS

    def write_script(self) -> None:
        """Write the commands collected by the last ``run()`` to ``assets/<session>/``.

        One script per group of parts (``<shard>.sh``: builds, then
        ``<shard>_merge.sh``: publication and cleanup), plus
        ``kmhelpers_apply.sh`` running them in order. Existing files are
        backed up with a ``.bak`` suffix.
        """
        if self._paths is None:
            logger.warning("Nothing planned: no script written")
            return
        self._script.write(self._paths.assets)

    def run(
        self,
        path: str,
        mode: ApplyMode,
        fail_on_error: bool = False,
        report_path: Optional[str] = None,
    ) -> ApplyResult:
        """Apply an index definition or span registry file to the kmindex registry.

        Builds every selected definition, then publishes every target.

        Args:
            path: Path to a YAML or JSON file containing either an
                ``IndexDefinition`` or a span registry.
            mode: Execution mode - controls whether to dry-run, plan, or apply.
            report_path: Where to write the YAML report, updated as the run
                goes (after each definition and each publish) and at the end.
            fail_on_error: Abort this run on the first build or merge error
                instead of continuing and returning ``PARTIAL``.

        Returns:
            An ``ApplyResult`` with the overall status and a per-index details
            dict. Status is ``SUCCESS`` when all operations complete without
            error, ``PARTIAL`` when some fail (only when
            ``fail_on_error=False``), or ``FAILED`` when a fatal error occurs.
        """
        path = os.path.realpath(path)
        self._script = ScriptRecorder(self.work_dir)
        self._paths = paths = BuildPaths.for_input(self.work_dir, path)
        report = RunReport(path, mode, self.config, report_path)
        report.set_build_dir(self.work_dir, paths.assets)
        try:
            self._run(report, path, mode, fail_on_error)
        finally:
            report.close()
        return report.result

    # ---
    # PRIVATE METHODS

    def _run(
        self, report: RunReport, path: str, mode: ApplyMode, fail_on_error: bool
    ) -> ApplyResult:
        paths = self._paths
        assert paths is not None
        idt = IndexDefinitionTools()

        input_type, data = detect_input(path, idt)
        report.set_input_type(input_type)
        if data is None or input_type is ApplyInputType.UNKNOWN:
            logger.error("Could not retrieve data type.")
            report.result.status = ApplyStatus.FAILED
            return report.result

        os.makedirs(paths.assets, exist_ok=True)
        builder = IndexBuilder(
            workdir=self.work_dir,
            registry_name=paths.registry,
            data_folder=paths.runs,
            log_folder=self.log_dir,
            assets_folder=os.path.relpath(paths.assets, self.work_dir),
        )
        try:
            plan = SOURCES[input_type](self.config, self._dbs).load(path, idt, data)
            targets = self._targets(plan)
            for name in targets:
                check_index_name(name)
        except Exception as e:
            Log.handle_exception(logger, e, f"Failed to load definition file '{path}'")
            report.result.status = ApplyStatus.FAILED
            return report.result

        layout = LayoutStore.next_to(path)
        executor = IndexExecutor(builder, mode, self.config, self._script, paths)
        samples = SampleResolver(
            self.config.sample_rootpath, mode.checks_files, self._dbs
        )

        own_targets = {t for t, parts in targets.items() if parts == [t]}
        built: dict[str, _Built] = {}
        for i in plan.definitions:
            assert i.name  # WorkPlan only holds named definitions
            logger.info(f"► Processing index definition '{i.name}'...")
            if i.name in own_targets and executor.has_index(i.name):
                logger.info(f"  └── {i.name} found in registry: skip")
                report.record(i.name, ApplyStatus.NONE)
                continue
            try:
                self._script.select(plan.script_of(i.name))
                b = self._build(
                    report, executor, samples, i, layout, plan.script_of(i.name)
                )
                if b:
                    built[i.name] = b
            except Exception as e:
                msg = f"   Failed to build index '{i.name}'"
                if report.fail(e, msg, i.name, fail_on_error):
                    return report.result
            report.dump()

        if self.config.filter_runs:
            # --run builds runs only: a later run with --shard alone merges them
            try:
                select_matching(
                    executor.run_names, self.config.filter_runs, "kmtricks run"
                )
            except ValueError as e:
                report.fail(e, "Invalid --run", None, True)
                return report.result
            logger.info("Runs built, nothing merged (--run)")
            return report.finish()

        for target, parts in targets.items():
            try:
                self._publish(report, executor, plan, target, parts, built)
            except Exception as e:
                msg = f"Failed to publish index '{target}'"
                if target in executor.publish_details:
                    report.record_merge(target, executor.publish_details[target])
                if report.fail(e, msg, target, fail_on_error):
                    return report.result
            report.dump()

        return report.finish()

    @staticmethod
    def _targets(plan: WorkPlan) -> dict[str, list[str]]:
        """Every index to publish, with the definitions it is made of.

        The merges of a span registry, plus every definition that is not a
        part of one of them (an index definition, or a part selected without
        its target), which is its own target.
        """
        targets = dict(plan.merges)
        in_merge = {p for parts in targets.values() for p in parts}
        for i in plan.definitions:
            if i.name and i.name not in in_merge:
                targets.setdefault(i.name, [i.name])
        return targets

    @staticmethod
    def _store_params(i: IndexDefinition, params: BuildParams) -> None:
        """Keep the size estimates of ``i`` in line with what is built."""
        i.partition_count = params.partitions

    def _resolve_params(
        self, i: IndexDefinition, sample_count: int, layout: LayoutStore
    ) -> BuildParams:
        params = resolve_build_params(i, sample_count, self.config, layout)
        # Keeps size estimates of i in line with what is built
        i.partition_count = params.partitions
        return params

    def _build(
        self,
        report: RunReport,
        executor: IndexExecutor,
        samples: SampleResolver,
        i: IndexDefinition,
        layout: LayoutStore,
        target: str,
    ) -> Optional[_Built]:
        """Build the runs of index ``i`` and record its outcome in ``report``."""
        # Partitions must be known before estimating the size
        params = self._resolve_params(i, i.sample_count, layout)
        size = i.get_stored_size()
        report.add_span_stats(i.span, i.sample_count, size.byte_count)
        logger.info(f"  └── Sample count: {i.sample_count}")
        logger.info(f"  └── Estimated build size: {size}")

        validate_definition(i)
        assert i.name is not None
        if not executor.claim(i.name):
            report.record(i.name, ApplyStatus.NONE)
            return None

        fof, issues = samples.build_fof(i)
        sample_count = fof.get_sample_count()
        if sample_count == 0:
            logger.warning(
                f"{indent_prefix(logger)}Skipping index '{i.name}' as no sample was added to it"
            )
            status = ApplyStatus.FAILED if issues else ApplyStatus.NONE
            report.record(i.name, status, issues=issues)
            return None

        # A part already in its target (rerun of a finished or interrupted session)
        published = executor.index_samples(target)
        overlap = published.intersection(fof.samples)
        if overlap and len(overlap) == sample_count:
            logger.info(
                f"  └── All samples of '{i.name}' are already in '{target}': skip"
            )
            report.record(i.name, ApplyStatus.NONE, issues=issues)
            return None
        if overlap:
            raise ValueError(
                f"{len(overlap)} of the {sample_count} samples of '{i.name}' are "
                f"already in '{target}' ({summarize_names(sorted(overlap))}): "
                f"a sample can only be added once"
            )

        # Samples may have been skipped since the estimate: resolve for the actual count
        if sample_count != i.sample_count:
            # i.partition_count now holds the first count: keep its source
            source = params.partitions_source
            params = self._resolve_params(i, sample_count, layout)
            params = replace(params, partitions_source=source)
        params = executor.align_with(target, params, i.name)
        self._store_params(i, params)
        report.record_params(i.name, i, params, sample_count)
        assert self._paths is not None
        report.record_script(i.name, os.path.join(self._paths.assets, f"{target}.sh"))
        try:
            runs = executor.build(i, fof, params)
        finally:
            report.record_runs(
                i.name,
                {
                    r: executor.run_details[r]
                    for r in executor.run_names
                    if r in executor.run_details
                    and (r == i.name or r.startswith(f"{i.name}_chunk"))
                },
            )
        report.record(i.name, ApplyStatus.SUCCESS, issues=issues)
        return _Built(i, params, runs)

    def _publish(
        self,
        report: RunReport,
        executor: IndexExecutor,
        plan: WorkPlan,
        target: str,
        parts: list[str],
        built: dict[str, _Built],
    ) -> None:
        """Publish the runs of ``parts`` as ``target`` and record the outcome.

        Parts that are indexes of ``BUILD_DIR`` (left by an older kmhelpers,
        which registered parts before merging them) are merged in as they are.
        A folder ``BUILD_DIR/<target>`` left unregistered by an interrupted
        publish, with nothing new to add, is registered.
        """
        new = [built[p] for p in parts if p in built]
        absorbed = [
            p
            for p in parts
            if p != target and (executor.has_index(p) or executor.has_folder(p))
        ]
        failed = [
            p
            for p in parts
            if p not in built
            and p not in absorbed
            and report.result_of(p) == ApplyStatus.FAILED.value
        ]
        if failed:
            logger.warning(
                f"Cannot publish '{target}': {summarize_names(failed)} failed to build"
            )
            if parts != [target]:
                report.fold_merge(target, parts)
            return
        if not new:
            if executor.has_folder(target) and not executor.has_index(target):
                logger.info(f"► Registering '{target}' (publish interrupted)...")
                self._script.select(plan.script_of(parts[0]))
                executor.register_folder(target, parts)
            self._record_publish(report, executor, plan, target, parts)
            return

        logger.info(f"► Publishing '{target}'...")
        runs = {name: n for b in new for name, n in b.runs.items()}
        self._script.select(plan.script_of(parts[0]))
        executor.publish(new[0].definition, target, runs, new[0].params, absorbed)
        self._record_publish(report, executor, plan, target, parts)

    def _record_publish(
        self,
        report: RunReport,
        executor: IndexExecutor,
        plan: WorkPlan,
        target: str,
        parts: list[str],
    ) -> None:
        assert self._paths is not None
        details = dict(executor.publish_details.get(target, {}))
        if details:
            group = ScriptRecorder.merge_group(plan.script_of(parts[0]))
            details["merge_script"] = os.path.join(self._paths.assets, f"{group}.sh")
            report.record_merge(target, details)
        if parts != [target]:
            report.fold_merge(target, parts)

    BuildParams dataclass

    Resolved kmindex build parameters for one index.

    partitions and minim_size are invariants of a shard: indexes are only mergeable when they share them. The minimizer size is fixed at design; the partition count by the design (-p) or by the first build of the shard (see resolve_partitions). chunk_size is set when the open-files ceiling can't fit every sample in one kmtricks build; the build is then split into chunks of at most that many samples, which are merged.

    partitions_source tells where the partition count comes from, for the report: design (compose -P), auto (sized from the limits), registered (the shard already built) or run (a run of the session already started). It does not take part in comparisons.

class BuildParams:
    """Resolved kmindex build parameters for one index.

    ``partitions`` and ``minim_size`` are invariants of a shard: indexes are
    only mergeable when they share them. The minimizer size is fixed at
    design; the partition count by the design (``-p``) or by the first build
    of the shard (see ``resolve_partitions``). ``chunk_size`` is set when the open-files ceiling can't fit
    every sample in one kmtricks build; the build is then split into chunks
    of at most that many samples, which are merged.

    ``partitions_source`` tells where the partition count comes from, for the
    report: ``design`` (``compose -P``), ``auto`` (sized from the limits),
    ``registered`` (the shard already built) or ``run`` (a run of the session
    already started). It does not take part in comparisons.
    """

    threads: int
    partitions: int
    chunk_size: Optional[int] = None
    minim_size: int = 10
    partitions_source: str = field(default="auto", compare=False)

    class RunReport:
    """Accumulates the outcome of one ``IndexOps.run`` into an ``ApplyResult``.

    ``details`` is organised by shard (``shards``): one entry per part, folded
    under its shard once published. With ``dump_path``, ``dump`` writes it as
    YAML, so a run still going or killed leaves its report so far.
    """

    def __init__(
        self,
        path: str,
        mode: ApplyMode,
        config: Optional[IndexOpsConfig] = None,
        dump_path: Optional[str] = None,
    ) -> None:
        wrapper = KmindexWrapper(dry_run=False)
        self._dump_path = dump_path
        self._start = time.monotonic()
        self.result = ApplyResult(
            status=ApplyStatus.NONE,
            input_type=ApplyInputType.UNKNOWN,
            mode=mode,
            details={
                "summary": {
                    "command": _COMMANDS[mode],
                    "status": RUNNING,
                    "input_type": ApplyInputType.UNKNOWN.value,
                    "started": datetime.now().isoformat(timespec="seconds"),
                },
                "input_file": path,
                "build_dir": {},
                "kmindex": {
                    "version": wrapper.kmindex_version(),
                    "path": wrapper.which,
                },
                "kmhelpers": _kmhelpers(),
                "system": _system(),
                "limits": _limits(config) if config else {},
                "shards": {},
            },
        )
        self._spans: dict = {}
        self._disk: dict[str, int] = {}
        if config:
            self.set_build_dir(config.workdir)

    @property
    def _shards(self) -> dict:
        shards: dict = self.result.details["shards"]
        return shards

    @property
    def _summary(self) -> dict:
        summary: dict = self.result.details["summary"]
        return summary

    def set_build_dir(self, build_dir: str, assets: Optional[str] = None) -> None:
        entry = {"path": build_dir, "logs": os.path.join(build_dir, "logs")}
        if assets:
            entry["assets"] = assets
        self.result.details["build_dir"] = entry

    def set_input_type(self, input_type: ApplyInputType) -> None:
        self.result.input_type = input_type
        self._summary["input_type"] = input_type.value

    def _entry(self, name: str) -> dict:
        entry: dict = self._shards.setdefault(name, {"result": NONE})
        return entry

    def record(
        self,
        name: str,
        status: ApplyStatus,
        error: str | None = None,
        issues: list[str] | None = None,
    ) -> None:
        """Set the result of ``name``, keeping what is already recorded for it."""
        entry = self._entry(name)
        entry["result"] = status.value
        if issues:
            entry["issues"] = issues
        if error:
            entry["error"] = error

    def record_params(self, name: str, definition, params, sample_count: int) -> None:
        """Parameters ``name`` is built with (``definition`` an ``IndexDefinition``,
        ``params`` its ``BuildParams``)."""
        entry = self._entry(name)
        entry["span"] = definition.span
        entry["samples"] = sample_count
        if sample_count != definition.sample_count:
            entry["planned_samples"] = definition.sample_count
        entry["params"] = {
            "threads": params.threads,
            "partitions": params.partitions,
            "partitions_source": params.partitions_source,
            "minim_size": params.minim_size,
            "chunk_size": params.chunk_size,
            "kmer_size": definition.kmer_size,
            "abundance_min": definition.abundance_min,
            "bf_size": definition.bf_size,
            "bitw": definition.bitw,
            "kmer_ratio": definition.kmer_ratio,
        }
        entry["est_size"] = str(definition.get_stored_size())

    def record_script(self, name: str, script: str) -> None:
        self._entry(name)["script"] = script

    def record_runs(self, name: str, runs: dict[str, dict]) -> None:
        """kmtricks runs of ``name``: run name -> what the executor knows of it."""
        threads = self._entry(name).get("params", {}).get("threads")
        ram = self.result.details["limits"].get("ram_bytes")
        entries = self._entry(name).setdefault("runs", {})
        for run, info in runs.items():
            info = dict(info)
            if "peak_rss_mb" in info:
                peak = info.pop("peak_rss_mb") * 1024 * 1024
                info["peak_rss"] = _size(peak)
                if ram:
                    info["ram_limit_use"] = round(peak / ram, 3)
            if threads and "max_cpu_pct" in info:
                info["threads"] = threads
                info["cpu_efficiency"] = round(info["max_cpu_pct"] / (100 * threads), 2)
            entries[run] = info

    def record_merge(self, target: str, info: dict) -> None:
        """How ``target`` was published (``IndexExecutor.publish_details``)."""
        info = dict(info)
        if "disk_bytes" in info:
            self._disk[target] = info["disk_bytes"]
            info["disk_size"] = _size(info.pop("disk_bytes"))
        self._entry(target).update(info)

    def add_span_stats(self, span: int, sample_count: int, byte_count: int) -> None:
        stats = self._spans.setdefault(span, {"shards": 0, "samples": 0, "bytes": 0})
        stats["shards"] += 1
        stats["samples"] += sample_count
        stats["bytes"] += byte_count

    def result_of(self, name: str) -> str | None:
        """Recorded result of ``name``, if any."""
        result: str | None = self._shards.get(name, {}).get("result")
        return result

    def fold_merge(self, target: str, parts: list[str]) -> None:
        """Move the entries of ``parts`` under ``target`` and derive its status."""
        entry: dict = {"result": SUCCESS}
        for part in parts:
            entry[part] = self._shards.pop(part, {"result": NONE})
        # publish details recorded before the fold stay on the target
        published = self._shards.pop(target, {})
        entry.update({k: v for k, v in published.items() if k != "result"})
        part_results = [v.get("result") for v in self._parts(entry)]
        entry["result"] = merge_status(part_results).value
        self._shards[target] = entry

    def fail(self, e: Exception, msg: str, key: str | None, abort: bool) -> bool:
        """Log an error and record it under ``key``. Returns True if the run must stop."""
        Log.handle_exception(logger, e, msg)
        if key:
            self.record(key, ApplyStatus.FAILED, Log.format_exception(e))
        self.result.status = ApplyStatus.FAILED if abort else ApplyStatus.PARTIAL
        self.dump()
        return abort

    def finish(self) -> ApplyResult:
        self.result.status = run_status(
            v.get("result") for v in self._shards.values() if isinstance(v, dict)
        )
        return self.result

    def close(self) -> ApplyResult:
        """Fill the summary with the final status, then dump the report."""
        self._fill_summary(final=True)
        self.dump()
        return self.result

    def dump(self) -> None:
        """Write the report to ``dump_path`` (no-op without one)."""
        if not self._dump_path:
            return
        self._fill_summary(final=False)
        try:
            with open(self._dump_path, "w") as f:
                yaml.dump(
                    self.result.details, f, default_flow_style=False, sort_keys=False
                )
        except OSError as e:
            logger.warning(f"Could not write the report {self._dump_path}: {e}")

    # ---
    # PRIVATE METHODS

    @staticmethod
    def _parts(entry: dict) -> list[dict]:
        """Part entries nested in a shard entry (dicts holding a ``result``)."""
        return [v for v in entry.values() if isinstance(v, dict) and "result" in v]

    def _fill_summary(self, final: bool) -> None:
        summary = self._summary
        if final:
            summary["status"] = self.result.status.value
            summary["finished"] = datetime.now().isoformat(timespec="seconds")
            summary["duration_s"] = round(time.monotonic() - self._start, 1)
        else:
            summary["duration_s"] = round(time.monotonic() - self._start, 1)

        counts = {s.value: 0 for s in ApplyStatus}
        runs = 0
        for entry in self._shards.values():
            result = entry.get("result", NONE)
            counts[result] = counts.get(result, 0) + 1
            for part in [entry] + self._parts(entry):
                runs += len(part.get("runs", {}))
        est = sum(s["bytes"] for s in self._spans.values())
        summary["counts"] = counts
        summary["totals"] = {
            "shards": len(self._shards),
            "samples": sum(s["samples"] for s in self._spans.values()),
            "runs": runs,
            "est_size": _size(est),
        }
        if self._disk:
            summary["totals"]["disk_size"] = _size(sum(self._disk.values()))
        summary["spans"] = {
            span: {
                "shards": s["shards"],
                "samples": s["samples"],
                "est_size": _size(s["bytes"]),
            }
            for span, s in sorted(self._spans.items())
        }
        # times next to the status, whatever was filled first
        order = ["command", "status", "input_type", "started", "finished", "duration_s"]
        ordered = {k: summary.pop(k) for k in order if k in summary}
        self.result.details["summary"] = {**ordered, **summary}

    def fold_merge(self, target: str, parts: list[str]) -> None:
    """Move the entries of ``parts`` under ``target`` and derive its status."""
    entry: dict = {"result": SUCCESS}
    for part in parts:
        entry[part] = self._shards.pop(part, {"result": NONE})
    # publish details recorded before the fold stay on the target
    published = self._shards.pop(target, {})
    entry.update({k: v for k, v in published.items() if k != "result"})
    part_results = [v.get("result") for v in self._parts(entry)]
    entry["result"] = merge_status(part_results).value
    self._shards[target] = entry

    def fail(self, e: Exception, msg: str, key: str | None, abort: bool) -> bool:
    """Log an error and record it under ``key``. Returns True if the run must stop."""
    Log.handle_exception(logger, e, msg)
    if key:
        self.record(key, ApplyStatus.FAILED, Log.format_exception(e))
    self.result.status = ApplyStatus.FAILED if abort else ApplyStatus.PARTIAL
    self.dump()
    return abort

    def run_status(results: Iterable[str | None]) -> ApplyStatus:
    """Overall status of a run from its per-item results."""
    entries = list(results)
    if not entries:
        return ApplyStatus.NONE
    has_failed = FAILED in entries
    has_ok = SUCCESS in entries
    if PARTIAL in entries or (has_failed and has_ok):
        return ApplyStatus.PARTIAL
    if has_failed:
        return ApplyStatus.FAILED
    if has_ok:
        return ApplyStatus.SUCCESS
    return ApplyStatus.NONE

    def merge_status(results: Iterable[str | None]) -> ApplyStatus:
    """Status of a merge target from the results of its parts.

    Unlike ``run_status``, a single failed part makes the merge PARTIAL,
    and a merge with no recorded part counts as SUCCESS.
    """
    entries = list(results)
    if entries and all(r == FAILED for r in entries):
        return ApplyStatus.FAILED
    if FAILED in entries:
        return ApplyStatus.PARTIAL
    if entries and all(r == NONE for r in entries):
        return ApplyStatus.NONE
    return ApplyStatus.SUCCESS

    def dir_bytes(path: str) -> int:
    """Bytes of the files under ``path`` (followed when it is a link)."""
    total = 0
    for root, _, files in os.walk(os.path.realpath(path)):
        for name in files:
            try:
                total += os.lstat(os.path.join(root, name)).st_size
            except OSError:
                pass
    return total

    def parse_kmtricks_log(path: str) -> dict:
    """What ``path`` tells about its run; empty when there is no log.

    Never raises: a partial log (failed run) gives only the fields it has.
    ``peak_rss_mb`` is the peak kmtricks reports itself, else the one sampled
    by the wrapper.
    """
    try:
        with open(path) as f:
            text = f.read()
    except OSError:
        return {}

    info: dict = {}
    for key, pattern in _HEADER.items():
        m = pattern.search(text)
        if m:
            info[key] = int(m.group(1)) if key == "return_code" else float(m.group(1))

    stamps = _STAMP.findall(text)
    if stamps:
        info["started"] = _iso(stamps[0])
        info["finished"] = _iso(stamps[-1])
    if m := _SAMPLES.search(text):
        info["samples"] = int(m.group(1))
        info["read_files"] = int(m.group(2))
    if m := _PARTITIONS.search(text):
        info["partitions"] = int(m.group(1))

    peak = _PEAK.search(text)
    if peak:
        info["peak_rss_mb"] = float(peak.group(1))
    elif "max_memory_mb" in info:
        info["peak_rss_mb"] = info["max_memory_mb"]
    info.pop("max_memory_mb", None)
    return info
